cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD waveshare_core2350b CACHE STRING "Target board")
set(PICO_PLATFORM rp2350-arm-s CACHE STRING "Platform")
set(PICO_NO_PICOTOOL 1 CACHE BOOL "Use the local UF2 packer")
if(NOT PICO_SDK_PATH)
    if(DEFINED ENV{PICO_SDK_PATH})
        set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
    else()
        get_filename_component(PICO_SDK_PATH "${CMAKE_CURRENT_LIST_DIR}/../external/pico-sdk" ABSOLUTE)
    endif()
endif()
if(NOT EXISTS "${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
    message(FATAL_ERROR "Fetch Pico SDK2.3.1 into external/pico-sdk or set PICO_SDK_PATH")
endif()
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(iigs_combined C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(iigs_combined main.c adb.c input.c)
target_include_directories(iigs_combined PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_compile_options(iigs_combined PRIVATE -Wall -Wextra -Werror -O3)
target_compile_definitions(iigs_combined PRIVATE PICO_STDIO_USB_STDOUT_TIMEOUT_US=5000)
target_link_libraries(iigs_combined pico_stdlib pico_multicore hardware_clocks hardware_sync hardware_pio hardware_dma hardware_watchdog)
pico_enable_stdio_usb(iigs_combined 1)
pico_enable_stdio_uart(iigs_combined 0)
pico_set_binary_type(iigs_combined copy_to_ram)
pico_add_bin_output(iigs_combined)
pico_add_hex_output(iigs_combined)
